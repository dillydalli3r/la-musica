# What the Android bundle installs into its site-packages.
#
# Same pins as the iOS bundle, with one difference: Pillow is NOT here. PyPI
# publishes no Android wheel for Pillow (checked 2026-09-19: pillow 12.3.0
# ships cp313/cp314/cp315 iOS wheels and no android_* file at all), so an
# Android build would have to compile Pillow and its C dependencies (zlib,
# libjpeg-turbo, freetype) for the NDK. The backend runs without it —
# mlo/deps.py exposes HAS_PIL and every image path in mlo/images.py checks it —
# so the Android bundle ships without Pillow and reports cover/resize features
# as unavailable instead of failing at import.
#
# pydantic-core is missing for the same reason and is NOT optional: FastAPI
# cannot import without it, so the pipeline refuses to stage an Android bundle
# whose site-packages lacks it (build it with cibuildwheel, see bundle.py).
fastapi==0.141.1
# The pydantic chain, pinned for the same reason as on iOS: pydantic-core has no
# Android wheel either, so the wheel this bundles is one we build (cibuildwheel).
pydantic==2.13.5
pydantic-core==2.46.5
uvicorn==0.52.4
websockets==17.1
python-multipart==0.0.32
aiofiles==25.1.0
httpx==0.28.1
